+static void __attribute__((unused)) apply_clahe_bgr888(uint8_t *img, int width, int height) {
+    const int GRID_X = 8;
+    const int GRID_Y = 8;
+    const int tile_w = width / GRID_X;   // 40px
+    const int tile_h = height / GRID_Y;  // 30px
+    const int tile_pixels = tile_w * tile_h; // 1200 pixels
+    const float clip_limit_factor = 1.5f;
+    const int clip_limit = (int)(clip_limit_factor * (tile_pixels / 256.0f)); // ~70
+
+    static uint8_t lut[GRID_Y][GRID_X][256];
+
+    // 1. T├¡nh to├ín Histogram & CDF cho tß╗½ng Tile 8x8
+    for (int gy = 0; gy < GRID_Y; gy++) {
+        for (int gx = 0; gx < GRID_X; gx++) {
+            int hist[256] = {0};
+            int start_x = gx * tile_w;
+            int start_y = gy * tile_h;
+
+            for (int y = 0; y < tile_h; y++) {
+                int px_y = start_y + y;
+                uint8_t *ptr = img + (px_y * width + start_x) * 3;
+                for (int x = 0; x < tile_w; x++) {
+                    uint8_t b = ptr[0];
+                    uint8_t g = ptr[1];
+                    uint8_t r = ptr[2];
+                    uint8_t lum = (uint8_t)((r * 77 + g * 150 + b * 29) >> 8);
+                    hist[lum]++;
+                    ptr += 3;
+                }
+            }
+
+            // Clip histogram ─æß╗â tr├ính khuß║┐ch ─æß║íi nhiß╗àu
+            int excess = 0;
+            for (int i = 0; i < 256; i++) {
+                if (hist[i] > clip_limit) {
+                    excess += (hist[i] - clip_limit);
+                    hist[i] = clip_limit;
+                }
+            }
+
+            // Ph├ón bß╗ò phß║ºn thß╗½a ─æß╗üu cho 256 bin
+            int bonus = excess / 256;
+            for (int i = 0; i < 256; i++) {
+                hist[i] += bonus;
+            }
+
+            // T├¡nh CDF (Cumulative Distribution Function)
+            int sum = 0;
+            float scale = 255.0f / tile_pixels;
+            for (int i = 0; i < 256; i++) {
+                sum += hist[i];
+                int val = (int)(sum * scale);
+                lut[gy][gx][i] = (val > 255) ? 255 : ((val < 0) ? 0 : val);
+            }
+        }
+    }
+
+    // 2. Nß╗Öi suy Bilinear sß╗æ nguy├¬n (Fixed Point 8-bit - Si├¬u nhanh ~2ms)
+    for (int y = 0; y < height; y++) {
+        int gy_fixed = (y * 256 / tile_h) - 128;
+        if (gy_fixed < 0) gy_fixed = 0;
+        int gy1 = gy_fixed >> 8;
+        int gy2 = gy1 + 1;
+        if (gy1 >= GRID_Y - 1) { gy1 = GRID_Y - 1; gy2 = GRID_Y - 1; }
+        int wy = gy_fixed & 255;
+        int inv_wy = 256 - wy;
+
+        for (int x = 0; x < width; x++) {
+            int gx_fixed = (x * 256 / tile_w) - 128;
+            if (gx_fixed < 0) gx_fixed = 0;
+            int gx1 = gx_fixed >> 8;
+            int gx2 = gx1 + 1;

+            int wx = gx_fixed & 255;
+            int inv_wx = 256 - wx;
+

+            uint8_t b = ptr[0];

+

+            int map12 = lut[gy1][gx2][lum];
+            int map21 = lut[gy2][gx1][lum];
+            int map22 = lut[gy2][gx2][lum];

+            if (lum > 0) {

+ * Vß║Ñn ─æß╗ü c┼⌐: handler block 4 gi├óy l├ám to├án bß╗Ö HTTP server chß║┐t.

+static void url_decode(char *dst, const char *src) {
+    char a = 0, b = 0;
+    while (*src) {
+        if ((*src == '%') &&
+            ((a = src[1]) && (b = src[2])) &&
+            (isxdigit((unsigned char)a) && isxdigit((unsigned char)b))) {
+            if (a >= 'a') a -= 'a' - 'A';
+            if (a >= 'A') a -= ('A' - 10);
+            else a -= '0';
+            if (b >= 'a') b -= 'a' - 'A';
+            if (b >= 'A') b -= ('A' - 10);
+            else b -= '0';
+            *dst++ = 16 * a + b;
+            src += 3;
+        } else if (*src == '+') {
+            *dst++ = ' ';
+            src++;
+        } else {
+            *dst++ = *src++;
+        }
+    }
+    *dst = '\0';
+}
